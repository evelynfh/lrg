// Receives historical-data upload emails and edit requests
const NOTIFY_EMAIL = 'your-email@example.com';

// Backup spreadsheet (see BACKUP at the bottom)
const BACKUP_SHEET_ID = 'YOUR_BACKUP_SPREADSHEET_ID';
const KEEP_WEEKLY_BACKUPS = 12;

// Species text stored for a whoosh session in which every bird escaped (only the firings are recorded)
const ZERO_CATCH_SPECIES = '(No birds caught)';

// Responses sheet columns (1-based):
// Date | Ringer | Species | Age | Extracted | Method | Processed | Fledged ringed | Pulli ringed |
// Broods assessed | Whoosh net active firings | Submitted at | Written-in species? | Written-in method?
const COL = {
  DATE: 1, RINGER: 2, SPECIES: 3, AGE: 4, EXTRACTED: 5, METHOD: 6, PROCESSED: 7,
  FLEDGED_RINGED: 8, PULLI_RINGED: 9, BROODS_ASSESSED: 10, WHOOSH_FIRINGS: 11,
  SUBMITTED_AT: 12, SPECIES_WRITE_IN: 13, METHOD_WRITE_IN: 14
};
const COL_COUNT = 14;

// Methods sheet columns: A = fledged methods, B = pullus methods. The Ringers sheet uses column A.
const METHODS_COL = { FLEDGED: 1, PULLUS: 2 };
const RINGERS_COL = 1;

// The endpoint is public, so nothing the browser says is trusted.
const LIMITS = {
  MAX_ENTRIES:          200,
  MAX_COUNT:            100000,
  MAX_TEXT:             120,
  MAX_FILES:            20,
  MAX_UPLOAD_BYTES:     20 * 1024 * 1024,
  MAX_NOTES:            2000,
  LOCK_WAIT_MS:         30000,
  DUPLICATE_WINDOW_S:   600,     // an identical submission inside this window is treated as a retry
  MAX_FLAG_COMMENT:     1000,
  FLAG_COOLDOWN_S:      600,     // one edit request per ringer and session date in this window
  MAX_FLAGS_PER_WINDOW: 20,      // all edit requests together, per FLAG_WINDOW_S
  FLAG_WINDOW_S:        21600
};

// Uploaded files with these extensions are refused
const BLOCKED_EXTENSIONS = [
  'exe', 'bat', 'cmd', 'com', 'scr', 'msi', 'jar', 'js', 'vbs', 'ps1', 'sh',
  'dll', 'apk', 'iso', 'html', 'htm', 'lnk', 'reg'
];

// ── Dashboard cache ──
// The dashboard needs, per ringer, their own sessions, plus two group-wide tallies (everyone's
// extractions by method and species, and everyone's processed birds by species) for "Species gaps".
// All three are built together from ONE pass over the Responses sheet and cached:
//   ringer_<name>            { sessions }
//   GROUP_FREQ_KEY           { method: { species: { ringer: count } } }
//   GROUP_PROCESSED_KEY      { species: { ringer: count } }
// The cache is rebuilt after every submission, every 5.5 hours by a self-rescheduling trigger
// (CacheService entries live at most 6 hours), and on demand by running refreshAllCaches().
const CACHE_TTL_S               = 21600;                 // CacheService's maximum (6 hours)
const CACHE_REFRESH_INTERVAL_MS = 5.5 * 60 * 60 * 1000;  // must stay under CACHE_TTL_S
const GROUP_FREQ_KEY            = 'group_freq_by_method';
const GROUP_PROCESSED_KEY       = 'group_processed';
const RINGER_INDEX_KEY          = 'ringer_index';        // list of the ringer_<name> keys currently cached

// ─────────────────────────────────────────────
// GET / POST
// ─────────────────────────────────────────────
function doGet(e) {
  try {
    const action = e.parameter.action;

    if (action === 'getTrainees') return jsonResponse(getTrainees());
    if (action === 'getMethods')  return jsonResponse(getMethods());

    if (action === 'getRingerData') {
      const ringer = squash_(e.parameter.ringer);
      if (!ringer) return jsonResponse({ error: 'Missing ringer parameter' });
      if (ringer.length > LIMITS.MAX_TEXT) return jsonResponse({ error: 'Ringer name is too long' });
      return jsonResponse(getRingerData(ringer));
    }

    return jsonResponse({ error: 'Unknown action' });
  } catch (err) {
    return errorResponse_(err);
  }
}

function doPost(e) {
  try {
    let data;
    try {
      data = JSON.parse(e.postData.contents);
    } catch (parseErr) {
      fail_('Invalid request.');
    }

    if (data && data.action === 'uploadFiles') return handleFileUpload(data);
    if (data && data.action === 'flagSession') return flagSession_(data);
    return submitSession_(data);
  } catch (err) {
    return errorResponse_(err);
  }
}

// ─────────────────────────────────────────────
// Session submission
// ─────────────────────────────────────────────
// Order of work: validate everything (nothing is written unless the whole submission is valid) →
// take the script lock → ignore an identical retry → resolve the ringer and methods to the spelling
// already on the sheet (adding them only if new) → write all rows in one call → refresh the cache → back up.
function submitSession_(data) {
  const clean = validateSubmission_(data);

  const lock = LockService.getScriptLock();
  try {
    lock.waitLock(LIMITS.LOCK_WAIT_MS);
  } catch (lockErr) {
    fail_('The server is busy — please try again in a moment.');
  }

  try {
    const cache = CacheService.getScriptCache();
    const dupKey = submissionFingerprint_(clean);
    if (cache.get(dupKey)) return jsonResponse({ result: 'success', duplicate: true });

    const ringersSheet = SS.getSheetByName('Ringers');
    const methodsSheet = SS.getSheetByName('Methods');
    const ringerMap    = loadCanonMap_(ringersSheet, RINGERS_COL);
    const fledgedMap   = loadCanonMap_(methodsSheet, METHODS_COL.FLEDGED);
    const pullusMap    = loadCanonMap_(methodsSheet, METHODS_COL.PULLUS);

    const ringer = canonicalise_(ringersSheet, RINGERS_COL, ringerMap, clean.trainee).value;
    const now = new Date();

    const rows = clean.entries.map(function (entry, index) {
      const isPullus = entry.age === 'Pullus';
      const method = isPullus
        ? canonicalise_(methodsSheet, METHODS_COL.PULLUS,  pullusMap,  entry.method)
        : canonicalise_(methodsSheet, METHODS_COL.FLEDGED, fledgedMap, entry.method);

      // Whoosh firings are a session-level number, stored on the FIRST row only
      // (the dashboard counts one value per submission — see sumWhooshFirings in dashboard.html)
      const firings = index === 0 && clean.whooshFirings !== null ? clean.whooshFirings : '';

      return [
        clean.date, ringer, entry.species, entry.age, entry.num, method.value,
        isPullus ? '' : entry.processed,
        isPullus ? '' : entry.fledgedRinged,
        isPullus ? entry.pulliRinged : '',
        isPullus ? entry.broodsAssessed : '',
        firings, now, entry.speciesWriteIn, method.added ? 'Yes' : 'No'
      ];
    });

    // Zero-catch whoosh session: one row holds the firings; species/age/method aren't needed
    if (!rows.length) {
      rows.push([clean.date, ringer, ZERO_CATCH_SPECIES, '', 0, '', '', '', '', '', clean.whooshFirings, now, 'No', 'No']);
    }

    const rSheet = SS.getSheetByName('Responses');
    rSheet.getRange(rSheet.getLastRow() + 1, 1, rows.length, COL_COUNT).setValues(rows);

    cache.put(dupKey, '1', LIMITS.DUPLICATE_WINDOW_S);

    // Neither of these may fail a submission that has already been written
    try { cacheAggregates_(buildAggregates_()); } catch (cacheErr) {
      Logger.log('Cache refresh warning: ' + cacheErr.message);
    }
    try { backupNewRows(rows, now); } catch (backupErr) {
      Logger.log('Backup warning: ' + backupErr.message);
    }

    return jsonResponse({ result: 'success' });
  } finally {
    lock.releaseLock();
  }
}

// ─────────────────────────────────────────────
// Validation
// ─────────────────────────────────────────────

// Throw an error whose message is safe to show to the person using the form.
function fail_(message) {
  const err = new Error(message);
  err.isValidation = true;
  throw err;
}

// Validation messages go back to the browser; anything unexpected is logged and a generic message returned.
function errorResponse_(err) {
  if (err && err.isValidation) return jsonResponse({ error: err.message });
  Logger.log('Unexpected error: ' + (err && err.stack ? err.stack : err));
  return jsonResponse({ error: 'Something went wrong on the server — please try again.' });
}

// Trim, collapse whitespace runs, drop control characters.
function squash_(value) {
  return String(value == null ? '' : value)
    .replace(/[\u0000-\u001f\u007f]/g, ' ')
    .replace(/\s+/g, ' ')
    .trim();
}

// Case-insensitive comparison key.
function keyOf_(value) {
  return squash_(value).toLowerCase();
}

// Required short text. Anything starting with = + - @ would be stored as a formula by Sheets.
function cleanText_(value, label) {
  const text = squash_(value);
  if (!text) fail_(label + ' is required.');
  if (text.length > LIMITS.MAX_TEXT) fail_(label + ' is too long.');
  if (/^[=+\-@]/.test(text)) fail_(label + ' cannot start with =, +, - or @.');
  return text;
}

// Whole number 0..MAX_COUNT. Missing/blank counts as 0.
function toCount_(value, label) {
  if (value === undefined || value === null || value === '') return 0;
  const n = Number(value);
  if (!Number.isInteger(n) || n < 0 || n > LIMITS.MAX_COUNT) {
    fail_(label + ' must be a whole number between 0 and ' + LIMITS.MAX_COUNT + '.');
  }
  return n;
}

// YYYY-MM-DD, a real calendar date, not in the future (a day and a half of slack for time zones).
function validateDate_(value) {
  const s = String(value == null ? '' : value).trim();
  const m = /^(\d{4})-(\d{2})-(\d{2})$/.exec(s);
  if (!m) fail_('Please provide a valid session date.');
  const y = Number(m[1]), mo = Number(m[2]), d = Number(m[3]);
  const t = new Date(Date.UTC(y, mo - 1, d));
  if (t.getUTCFullYear() !== y || t.getUTCMonth() !== mo - 1 || t.getUTCDate() !== d || y < 1950) {
    fail_('Please provide a valid session date.');
  }
  if (t.getTime() > Date.now() + 36 * 3600 * 1000) fail_('The session date cannot be in the future.');
  return s;
}

// Returns a cleaned copy of the submission, or throws via fail_(). Writes nothing.
function validateSubmission_(data) {
  if (!data || typeof data !== 'object') fail_('Invalid request.');

  const trainee = cleanText_(data.trainee, 'Ringer name');
  const date    = validateDate_(data.date);

  let whooshFirings = null;
  if (data.whooshFirings !== null && data.whooshFirings !== undefined && data.whooshFirings !== '') {
    whooshFirings = toCount_(data.whooshFirings, 'Whoosh net firings');
  }

  // No species entries is allowed only for a zero-catch whoosh session
  if (!Array.isArray(data.entries)) fail_('No entries in submission.');
  if (!data.entries.length && !(whooshFirings > 0)) fail_('No entries in submission.');
  if (data.entries.length > LIMITS.MAX_ENTRIES) fail_('Too many entries in one submission.');

  const entries = data.entries.map(function (raw, i) {
    const where = 'Entry ' + (i + 1);
    if (!raw || typeof raw !== 'object') fail_(where + ' is invalid.');
    if (raw.age !== 'Fledged' && raw.age !== 'Pullus') fail_(where + ': age must be Fledged or Pullus.');
    const isPullus = raw.age === 'Pullus';

    const species = cleanText_(raw.species, where + ' species');
    const method  = cleanText_(raw.method,  where + ' method');
    const num     = toCount_(raw.num, where + ' extracted');

    // Fields that don't apply to the age class are forced to 0, whatever the browser sent
    const processed      = isPullus ? 0 : toCount_(raw.processed,     where + ' processed');
    const fledgedRinged  = isPullus ? 0 : toCount_(raw.fledgedRinged, where + ' ringed');
    const pulliRinged    = isPullus ? toCount_(raw.pulliRinged,    where + ' pulli ringed')    : 0;
    const broodsAssessed = isPullus ? toCount_(raw.broodsAssessed, where + ' broods assessed') : 0;

    if (!isPullus && fledgedRinged > processed) fail_(where + ': ringed cannot be more than processed.');
    if (!isPullus && num === 0 && processed === 0) fail_(where + ': needs at least one extracted or processed bird.');
    if (isPullus && pulliRinged === 0 && broodsAssessed === 0) fail_(where + ': needs at least one pulli ringed or brood assessed.');

    return {
      species, age: raw.age, num, method, processed, fledgedRinged, pulliRinged, broodsAssessed,
      speciesWriteIn: raw.speciesWriteIn === 'Yes' ? 'Yes' : 'No'
    };
  });

  return { trainee, date, whooshFirings, entries };
}

// Stable key for "this exact submission", used to ignore a retry after a lost response.
function submissionFingerprint_(clean) {
  const raw = JSON.stringify([clean.trainee, clean.date, clean.whooshFirings, clean.entries]);
  const digest = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, raw, Utilities.Charset.UTF_8);
  return 'dup_' + Utilities.base64EncodeWebSafe(digest);
}

// ─────────────────────────────────────────────
// Ringer / method names: one canonical spelling
// ─────────────────────────────────────────────

// lower-cased name -> the spelling stored on the sheet (first one wins)
function loadCanonMap_(sheet, colIndex) {
  const map = {};
  getColumnValues(sheet, colIndex).forEach(function (v) {
    const k = keyOf_(v);
    if (!(k in map)) map[k] = v;
  });
  return map;
}

// Return the stored spelling of `raw` if it exists (ignoring case and extra spaces);
// otherwise add it to the sheet column and remember it in `map`.
function canonicalise_(sheet, colIndex, map, raw) {
  const text = squash_(raw);
  const k = text.toLowerCase();
  if (k in map) return { value: map[k], added: false };
  appendToColumn_(sheet, colIndex, text);
  map[k] = text;
  return { value: text, added: true };
}

// Write into the first empty cell below the header in one column, so each column of the
// Methods sheet fills independently of the other.
function appendToColumn_(sheet, colIndex, value) {
  const maxRows = sheet.getMaxRows();
  const values = sheet.getRange(1, colIndex, maxRows, 1).getValues();
  let row = -1;
  for (let i = 1; i < values.length; i++) {          // i = 1 skips the header
    if (!String(values[i][0]).trim()) { row = i + 1; break; }
  }
  if (row === -1) {
    sheet.insertRowAfter(maxRows);
    row = maxRows + 1;
  }
  sheet.getRange(row, colIndex).setValue(value);
}

// ─────────────────────────────────────────────
// Edit requests: a ringer flags one of their sessions as wrong
// ─────────────────────────────────────────────
// Expects { action: 'flagSession', ringer, date, comment } and emails NOTIFY_EMAIL.
// The ringer and session must exist, and requests are rate-limited, because every one sends an email.
function flagSession_(data) {
  if (!data || typeof data !== 'object') fail_('Invalid request.');
  const ringerIn = cleanText_(data.ringer, 'Ringer name');
  const date     = validateDate_(data.date);
  const comment  = String(data.comment == null ? '' : data.comment)
    .replace(/[\u0000-\u0008\u000b\u000c\u000e-\u001f\u007f]/g, '')
    .trim();
  if (comment.length < 5) fail_('Please say what needs changing.');
  if (comment.length > LIMITS.MAX_FLAG_COMMENT) fail_('The comment is too long.');

  const ringer = loadCanonMap_(SS.getSheetByName('Ringers'), RINGERS_COL)[keyOf_(ringerIn)];
  if (!ringer) fail_('That ringer could not be found.');

  // Find that ringer's rows for that date
  const rSheet = SS.getSheetByName('Responses');
  const lastRow = rSheet.getLastRow();
  const submitted = {};
  let rowCount = 0;
  if (lastRow >= 2) {
    rSheet.getRange(2, 1, lastRow - 1, COL.SUBMITTED_AT).getValues().forEach(function (row) {
      if (keyOf_(row[COL.RINGER - 1]) === keyOf_(ringer) && formatDate_(row[COL.DATE - 1]) === date) {
        rowCount++;
        const ts = formatTimestamp_(row[COL.SUBMITTED_AT - 1]);
        if (ts) submitted[ts] = true;
      }
    });
  }
  if (!rowCount) fail_('That session could not be found.');

  // Rate limits (best effort: the cache can be cleared at any time)
  const cache = CacheService.getScriptCache();
  const sessionKey = 'flag_' + keyOf_(ringer).replace(/\s+/g, '_') + '_' + date;
  if (cache.get(sessionKey)) fail_('A request for this session was sent a moment ago. Please wait a few minutes before sending another.');
  const used = Number(cache.get('flag_window')) || 0;
  if (used >= LIMITS.MAX_FLAGS_PER_WINDOW) fail_('Too many edit requests have been sent recently. Please try again later.');
  if (MailApp.getRemainingDailyQuota() < 1) fail_('The email limit for today has been reached. Please try again tomorrow.');

  const timestamps = Object.keys(submitted).sort();
  MailApp.sendEmail({
    to:      NOTIFY_EMAIL,
    subject: 'Ringing Tracker \u2013 edit request from ' + ringer + ' (session ' + date + ')',
    body: [
      'A ringer has asked for one of their sessions to be edited.',
      '',
      'Ringer: ' + ringer,
      'Date of entry (session date): ' + date,
      'Rows for that session: ' + rowCount,
      'Submitted at: ' + (timestamps.length ? timestamps.join(', ') : '(no timestamp recorded)'),
      'Edit requested on: ' + Utilities.formatDate(new Date(), 'Europe/London', 'yyyy-MM-dd HH:mm') + ' (Europe/London)',
      '',
      'Comment:',
      comment,
      '',
      'Spreadsheet: ' + SS.getUrl()
    ].join('\n')
  });

  cache.put(sessionKey, '1', LIMITS.FLAG_COOLDOWN_S);
  cache.put('flag_window', String(used + 1), LIMITS.FLAG_WINDOW_S);
  return jsonResponse({ result: 'success' });
}

// ─────────────────────────────────────────────
// File upload
// ─────────────────────────────────────────────
// Expects { action: 'uploadFiles', sender, notes, files: [{ name, mimeType, data: <base64> }] }
// and emails the files to NOTIFY_EMAIL (Gmail's limit is 25 MB; the page and this function cap it at 20 MB).
function handleFileUpload(data) {
  if (!Array.isArray(data.files) || !data.files.length) fail_('No files received.');
  if (data.files.length > LIMITS.MAX_FILES) fail_('Too many files in one upload.');

  const sender = squash_(data.sender || 'Unknown ringer').slice(0, 100) || 'Unknown ringer';
  const notes  = String(data.notes == null ? '' : data.notes)
    .replace(/[\u0000-\u0008\u000b\u000c\u000e-\u001f\u007f]/g, '')
    .trim()
    .slice(0, LIMITS.MAX_NOTES);

  if (MailApp.getRemainingDailyQuota() < 1) {
    fail_('The upload limit for today has been reached — please try again tomorrow.');
  }

  let totalBytes = 0;
  const attachments = data.files.map(function (f) {
    if (!f || typeof f.data !== 'string') fail_('One of the files could not be read.');

    const name = squash_(String(f.name || 'file').replace(/[\\/]/g, '_')).slice(0, 150) || 'file';
    const ext = name.indexOf('.') >= 0 ? name.split('.').pop().toLowerCase() : '';
    if (BLOCKED_EXTENSIONS.indexOf(ext) !== -1) fail_('Files of type .' + ext + ' cannot be uploaded.');

    let bytes;
    try {
      bytes = Utilities.base64Decode(f.data);
    } catch (decodeErr) {
      fail_('One of the files could not be read.');
    }
    totalBytes += bytes.length;
    if (totalBytes > LIMITS.MAX_UPLOAD_BYTES) fail_('The files are larger than the 20 MB limit.');

    return Utilities.newBlob(bytes, String(f.mimeType || 'application/octet-stream').slice(0, 100), name);
  });

  const n = attachments.length;
  MailApp.sendEmail({
    to:      NOTIFY_EMAIL,
    subject: 'Ringing Tracker \u2013 historical data from ' + sender + ' (' + n + ' file' + (n !== 1 ? 's' : '') + ')',
    body: [
      'Historical ringing data uploaded via Ringing Tracker.',
      '',
      'Submitted by: ' + sender,
      'Submitted at: ' + new Date().toLocaleString('en-GB', { timeZone: 'Europe/London' }),
      '',
      'Files (' + n + '):',
      attachments.map(function (b) { return '  \u2022 ' + b.getName(); }).join('\n'),
      '',
      'Notes:',
      notes || '(none)'
    ].join('\n'),
    attachments: attachments
  });

  return jsonResponse({ result: 'success', count: n });
}

// ─────────────────────────────────────────────
// Dashboard data (cached — see the note in CONFIGURATION)
// ─────────────────────────────────────────────

function ringerCacheKey_(name) {
  return 'ringer_' + keyOf_(name).replace(/\s+/g, '_');
}

// One pass over the Responses sheet. Returns everything the cache holds:
//   sessionsByKey   { ringerCacheKey: [{ date, entries[] }] }   (dates ascending)
//   groupFreqByMethod   { method: { species: { ringer: count } } }   (extracted birds, all ringers)
//   groupProcessedFreq  { species: { ringer: count } }               (processed birds, all ringers)
function buildAggregates_() {
  const sheet = SS.getSheetByName('Responses');
  const lastRow = sheet.getLastRow();
  const rows = lastRow < 2 ? [] : sheet.getRange(2, 1, lastRow - 1, COL.SUBMITTED_AT).getValues();

  const byRinger = {};              // cache key -> { date -> entries[] }
  const groupFreqByMethod = {};
  const groupProcessedFreq = {};

  rows.forEach(function (row) {
    const date    = formatDate_(row[COL.DATE - 1]);
    const ringer  = squash_(row[COL.RINGER - 1]);
    const species = squash_(row[COL.SPECIES - 1]);
    if (!ringer || !species || !date) return;

    const extracted = Number(row[COL.EXTRACTED - 1]) || 0;
    const method    = squash_(row[COL.METHOD - 1]);
    const processed = Number(row[COL.PROCESSED - 1]) || 0;
    const rawFirings = row[COL.WHOOSH_FIRINGS - 1];

    const key = ringerCacheKey_(ringer);
    if (!byRinger[key]) byRinger[key] = {};
    if (!byRinger[key][date]) byRinger[key][date] = [];
    byRinger[key][date].push({
      species: species,
      age: squash_(row[COL.AGE - 1]),
      extracted: extracted,
      method: method,
      processed: processed,
      fledgedRinged:  Number(row[COL.FLEDGED_RINGED - 1]) || 0,
      pulliRinged:    Number(row[COL.PULLI_RINGED - 1]) || 0,
      broodsAssessed: Number(row[COL.BROODS_ASSESSED - 1]) || 0,
      whooshFirings:  (rawFirings === '' || rawFirings === null || isNaN(Number(rawFirings))) ? null : Number(rawFirings),
      // Every row of one submission shares this timestamp, so (date + submittedAt) identifies a submission
      submittedAt: formatTimestamp_(row[COL.SUBMITTED_AT - 1]),
      zeroCatch: species === ZERO_CATCH_SPECIES
    });

    if (method && extracted > 0) {
      if (!groupFreqByMethod[method]) groupFreqByMethod[method] = {};
      if (!groupFreqByMethod[method][species]) groupFreqByMethod[method][species] = {};
      groupFreqByMethod[method][species][ringer] = (groupFreqByMethod[method][species][ringer] || 0) + extracted;
    }
    if (processed > 0) {
      if (!groupProcessedFreq[species]) groupProcessedFreq[species] = {};
      groupProcessedFreq[species][ringer] = (groupProcessedFreq[species][ringer] || 0) + processed;
    }
  });

  const sessionsByKey = {};
  Object.keys(byRinger).forEach(function (key) {
    sessionsByKey[key] = Object.keys(byRinger[key]).sort().map(function (date) {
      return { date: date, entries: byRinger[key][date] };
    });
  });

  return { sessionsByKey: sessionsByKey, groupFreqByMethod: groupFreqByMethod, groupProcessedFreq: groupProcessedFreq };
}

// Store one cache entry. If it is too big for CacheService (100 KB per key) it is simply skipped,
// and the next read of it falls back to a fresh pass over the sheet.
function cachePut_(key, value) {
  const cache = CacheService.getScriptCache();
  try {
    cache.put(key, JSON.stringify(value), CACHE_TTL_S);
  } catch (e) {
    cache.remove(key);   // never leave an older, now-stale entry behind
    Logger.log('Cache entry not stored (' + key + '): ' + e.message);
  }
}

// Writes every entry, and removes entries for ringers that no longer appear (e.g. rows deleted by hand).
// CacheService can't list its keys, so the keys written last time are kept in RINGER_INDEX_KEY.
// Every ringer on the Ringers sheet gets an entry, empty if they have no rows yet.
// Caller must hold the script lock.
function cacheAggregates_(agg) {
  const cache = CacheService.getScriptCache();

  const sessionsByKey = Object.assign({}, agg.sessionsByKey);
  getColumnValues(SS.getSheetByName('Ringers'), RINGERS_COL).forEach(function (name) {
    const key = ringerCacheKey_(name);
    if (!sessionsByKey[key]) sessionsByKey[key] = [];
  });
  const keys = Object.keys(sessionsByKey);

  let previous = [];
  try { previous = JSON.parse(cache.get(RINGER_INDEX_KEY) || '[]'); } catch (e) {}
  const stale = previous.filter(function (k) { return !(k in sessionsByKey); });
  for (let i = 0; i < stale.length; i += 100) cache.removeAll(stale.slice(i, i + 100));   // 100 keys per call

  cachePut_(GROUP_FREQ_KEY, agg.groupFreqByMethod);
  cachePut_(GROUP_PROCESSED_KEY, agg.groupProcessedFreq);
  keys.forEach(function (key) { cachePut_(key, { sessions: sessionsByKey[key] }); });
  cachePut_(RINGER_INDEX_KEY, keys);
}

function getRingerData(ringer) {
  const key = ringerCacheKey_(ringer);
  const hit = CacheService.getScriptCache().getAll([key, GROUP_FREQ_KEY, GROUP_PROCESSED_KEY]);

  if (hit[key] && hit[GROUP_FREQ_KEY] && hit[GROUP_PROCESSED_KEY]) {
    try {
      return {
        sessions: JSON.parse(hit[key]).sessions,
        groupFreqByMethod: JSON.parse(hit[GROUP_FREQ_KEY]),
        groupProcessedFreq: JSON.parse(hit[GROUP_PROCESSED_KEY])
      };
    } catch (e) { /* a corrupt entry: fall through and rebuild */ }
  }

  // Something is missing: rebuild and re-cache everything from one pass. This happens under the script
  // lock so a submission can't land between reading the sheet and writing the cache. If the lock can't be
  // had within 10s the data is still returned, just not cached.
  const lock = LockService.getScriptLock();
  let locked = false;
  try { locked = lock.tryLock(10000); } catch (e) {}
  try {
    const agg = buildAggregates_();
    const sessions = agg.sessionsByKey[key] || [];
    if (locked) {
      cacheAggregates_(agg);
      if (!agg.sessionsByKey[key]) cachePut_(key, { sessions: [] });   // a ringer with no rows yet still gets a (warm) entry
    }
    return { sessions: sessions, groupFreqByMethod: agg.groupFreqByMethod, groupProcessedFreq: agg.groupProcessedFreq };
  } finally {
    if (locked) lock.releaseLock();
  }
}

// ─────────────────────────────────────────────
// Keeping the cache warm
// ─────────────────────────────────────────────
// Rebuilds the whole cache and books the next rebuild CACHE_REFRESH_INTERVAL_MS from now (Apps Script's
// fixed-interval triggers only allow 1/2/4/6/8/12 hours, so this reschedules itself). Without it a
// once-a-day visitor would always find an expired cache.
//
// Run it once by hand from the editor to start the chain. Run it again any time after editing the
// Responses sheet directly (the form rebuilds the cache itself, but manual edits are only seen at the next
// rebuild). It is safe to re-run: it replaces its own trigger, so there is only ever one.
function refreshAllCaches() {
  try {
    const lock = LockService.getScriptLock();
    lock.waitLock(LIMITS.LOCK_WAIT_MS);
    try { cacheAggregates_(buildAggregates_()); } finally { lock.releaseLock(); }
  } catch (err) {
    Logger.log('refreshAllCaches failed: ' + (err && err.stack ? err.stack : err));
  } finally {
    // Always book the next run, even after an error, so one failure can't end the chain
    ScriptApp.getProjectTriggers()
      .filter(function (t) { return t.getHandlerFunction() === 'refreshAllCaches'; })
      .forEach(function (t) { ScriptApp.deleteTrigger(t); });
    ScriptApp.newTrigger('refreshAllCaches').timeBased().after(CACHE_REFRESH_INTERVAL_MS).create();
  }
}

// Non-empty values below the header in one column (1-based column index)
function getColumnValues(sheet, colIndex) {
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return [];
  return sheet.getRange(2, colIndex, lastRow - 1, 1)
    .getValues()
    .flat()
    .map(function (v) { return String(v).trim(); })
    .filter(Boolean);
}

// Dates are read in the SPREADSHEET's time zone, because that is the zone Sheets used when it turned the
// stored "YYYY-MM-DD" text into a date. Set both to Europe/London: File > Settings in the sheet, and
// Project Settings in the Apps Script editor.
function formatDate_(value) {
  if (!value) return '';
  const tz = SS.getSpreadsheetTimeZone();
  if (Object.prototype.toString.call(value) === '[object Date]') {
    return isNaN(value) ? '' : Utilities.formatDate(value, tz, 'yyyy-MM-dd');
  }
  const s = String(value).trim();
  if (/^\d{4}-\d{2}-\d{2}$/.test(s)) return s;
  const d = new Date(s);
  return isNaN(d) ? s : Utilities.formatDate(d, tz, 'yyyy-MM-dd');
}

// "Submitted at" as text, e.g. "2026-11-03 14:07:52" (empty if blank)
function formatTimestamp_(value) {
  if (!value) return '';
  if (Object.prototype.toString.call(value) === '[object Date]') {
    return isNaN(value) ? '' : Utilities.formatDate(value, SS.getSpreadsheetTimeZone(), 'yyyy-MM-dd HH:mm:ss');
  }
  return String(value).trim();
}

function jsonResponse(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

// ═════════════════════════════════════════════
// BACKUP
// ═════════════════════════════════════════════
// Per submission: backupNewRows() appends the new rows to a monthly tab ("Sessions 2026-03") in the backup
// spreadsheet. Weekly: backupResponses() copies Responses, Ringers and Methods to dated snapshot tabs
// and keeps the newest KEEP_WEEKLY_BACKUPS of each (the monthly tabs are never deleted).
//
// Setup (once): create a backup spreadsheet, paste its ID into BACKUP_SHEET_ID, then run
// setupBackupTrigger() from the editor and accept the permissions popup.

// Registers the weekly run (Mondays at 07:00). Safe to re-run.
function setupBackupTrigger() {
  ScriptApp.getProjectTriggers()
    .filter(function (t) { return t.getHandlerFunction() === 'backupResponses'; })
    .forEach(function (t) { ScriptApp.deleteTrigger(t); });

  ScriptApp.newTrigger('backupResponses')
    .timeBased()
    .onWeekDay(ScriptApp.WeekDay.MONDAY)
    .atHour(7)
    .create();

  Logger.log('Backup trigger registered — will run every Monday at 07:00.');
}

// Called inside the lock in submitSession_, so two submissions can't append to the monthly tab at once.
function backupNewRows(rows, now) {
  if (!BACKUP_SHEET_ID) return;

  const backupSS = SpreadsheetApp.openById(BACKUP_SHEET_ID);
  const tabName = 'Sessions ' + formatDate_(now).slice(0, 7);
  let tab = backupSS.getSheetByName(tabName);

  if (!tab) {
    tab = backupSS.insertSheet(tabName);
    const header = SS.getSheetByName('Responses').getRange(1, 1, 1, COL_COUNT).getValues();
    tab.getRange(1, 1, 1, COL_COUNT).setValues(header);
    tab.setFrozenRows(1);
  }

  tab.getRange(tab.getLastRow() + 1, 1, rows.length, COL_COUNT).setValues(rows);
}

// Runs weekly, and can be run by hand at any time.
function backupResponses() {
  if (!BACKUP_SHEET_ID) throw new Error('Backup aborted: please set BACKUP_SHEET_ID before running.');

  const backupSS = SpreadsheetApp.openById(BACKUP_SHEET_ID);
  const today = formatDate_(new Date());

  // [source sheet name, snapshot tab prefix]
  [['Responses', 'Backup'], ['Ringers', 'Ringers'], ['Methods', 'Methods']].forEach(function (pair) {
    const source = SS.getSheetByName(pair[0]);
    if (!source) { Logger.log('Backup skipped: no sheet called "' + pair[0] + '".'); return; }

    const tabName = pair[1] + ' ' + today;
    const rowsWritten = snapshotSheet_(backupSS, source, tabName);
    if (rowsWritten < 0) {
      Logger.log('Backup skipped: "' + pair[0] + '" is empty.');
    } else {
      Logger.log('Backup complete: ' + rowsWritten + ' rows written to "' + tabName + '".');
      pruneBackupTabs_(backupSS, pair[1], KEEP_WEEKLY_BACKUPS);
    }
  });
}

// Copy one sheet's values into a fresh tab (replacing a same-day tab). Returns the number of data rows
// written, or -1 if the source sheet is empty.
function snapshotSheet_(backupSS, source, tabName) {
  const lastRow = source.getLastRow();
  const lastCol = source.getLastColumn();
  if (lastRow < 1 || lastCol < 1) return -1;

  const data = source.getRange(1, 1, lastRow, lastCol).getValues();

  const existing = backupSS.getSheetByName(tabName);
  if (existing) backupSS.deleteSheet(existing);

  const tab = backupSS.insertSheet(tabName);
  tab.getRange(1, 1, data.length, data[0].length).setValues(data);
  tab.setFrozenRows(1);
  tab.autoResizeColumns(1, data[0].length);

  backupSS.setActiveSheet(tab);
  backupSS.moveActiveSheet(1);
  return lastRow - 1;
}

// Keep only the newest `keep` tabs named "<prefix> YYYY-MM-DD"; other tabs are never touched.
function pruneBackupTabs_(backupSS, prefix, keep) {
  const pattern = new RegExp('^' + prefix + ' \\d{4}-\\d{2}-\\d{2}$');
  backupSS.getSheets()
    .filter(function (s) { return pattern.test(s.getName()); })
    .sort(function (a, b) { return b.getName().localeCompare(a.getName()); })   // newest first
    .slice(keep)
    .forEach(function (s) { backupSS.deleteSheet(s); });
}
